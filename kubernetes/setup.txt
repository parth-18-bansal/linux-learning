// Generate Certificat Authority cert and private key
____________________________________________________________________________________________________________
1. this is to generate the private RSA key, it is used by CA for siging the ssl certificate.
-> openssl genrsa -out ca.key 2048

2. this is to generate a CA certificate. it is like an identity card of CA, we stores this cert in the
client's trusted CA list, by this clients can check server's cert that server's SSL cert is signed by
trusted CA or not.
-> openssl req -x509 -new -nodes -key ca.key -subj "/CN=kubernetes-ca" -days 3650 -out ca.crt

3. To checks the details of the cert
-> openssl x509 -in ca.crt -text -noout


** Store this key and cert somewhere in the system, we later use these to sign the SSL certificates.

// Generate tls certificate for apiserver
___________________________________________________________________________________________________________
1. generate the rsa private key
-> openssl genrsa -out apiserver.key 2048

2. write the apiserver.cnf files, it stores the details like for which ip, dns this certificate is valid
and this file is used to create the CSR(certificate request), CSR contains the public key of the apiserver
public key can be derieved from private key openssl derive it automatically and add in the CSR and cert

-> openssl req -new -key apiserver.key -out apiserver.csr -config apiserver.cnf

3. generate the signed tls certificate: it contains the signature and public key of apiserver
-> openssl x509 -req -in apiserver.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out apiserver.crt -days 365 -extensions v3_req -extfile apiserver.cnf

4. check the cert details:
-> openssl x509 -in apiserver.crt -text -noout



// Generate service account public and private key
_________________________________________________________________________________________________
1. Generate the private key
openssl genrsa -out sa.key 2048

2. extract the public key from private key
openssl rsa -in sa.key -pubout -out sa.pub

[Unit]
Description=Kubernetes API Server
After=network.target

[Service]
ExecStart=/usr/local/bin/kube-apiserver \
  --advertise-address= \
  --bind-address=0.0.0.0 \
  --secure-port=6443 \
  --allow-privileged=true \
  --enable-admission-plugins=NodeRestriction \
  --authorization-mode=Node,RBAC \
  --tls-cert-file=/home/parth/kubernetes/apiserver-tls-certs/apiserver.crt \
  --tls-private-key-file=/home/parth/kubernetes/apiserver-tls-certs/apiserver.key \
  --client-ca-file=/home/parth/kubernetes/kubernetes-ca/ca.crt \
  --service-account-issuer=https://kubernetes.default.svc.cluster.local \
  --service-account-key-file=/home/parth/kubernetes/sa-keys/sa.pub \
  --service-account-signing-key-file=/home/parth/kubernetes/sa-keys/sa.key \
  --etcd-servers=http://127.0.0.1:2379 \
  --service-cluster-ip-range=10.96.0.0/12 \
  --enable-bootstrap-token-auth=true \
  --v=2

Restart=on-failure
RestartSec=5

[Install]
WantedBy=multi-user.target



sudo systemctl daemon-reload

// Download and install the kubectl
______________________________________________________________________________________________________
1. download the binary; curl -LO https://dl.k8s.io/release/v1.36.0/bin/linux/amd64/kubectl

2. check the checksum
curl -LO https://dl.k8s.io/release/v1.36.0/bin/linux/amd64/kubectl.sha256
echo "$(cat kubectl.sha256)  kubectl" | sha256sum --check

3. install the binary(copy the binary in usr/local/bin)
sudo install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl

4. check the version
kubectl version --client



// connect kubectl and apiserver
______________________________________________________________________________________________________
1. generate private key and cert for kubectl because in connection we will use mTLS
-> openssl genrsa -out admin.key 2048
   openssl req -new -key admin.key -out admin.csr -subj "/CN=admin/O=system:masters"
   openssl x509 -req -in admin.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out admin.crt -days 365

2. generate the kubeconfig file
  a. set cluster, trusted CA
    kubectl config set-cluster my-cluster --server=https://127.0.0.1:6443 --certificate-authority=ca.crt --embed-certs=true --kubeconfig=admin.kubeconfig
  b. set client cert and key
    kubectl config set-credentials admin --client-certificate=admin.crt --client-key=admin.key --embed-certs=true --kubeconfig=admin.kubeconfig
  c. set and use context
    kubectl config set-context default --cluster=my-cluster --user=admin --kubeconfig=admin.kubeconfig
    kubectl config use-context default --kubeconfig=admin.kubeconfig

3. after generating the admin.kubeconfig paste it in ~/.kube/config file
-> cp admin.kubeconfig ~/.kube/config 

and set permission to 600

// generate key and tls certificate for kube-controller-manager
________________________________________________________________________________________________________
1. define what to put in the cert using kcm.cnf file

2. generate the private key
->  openssl genrsa -out kube-controller-manager.key 2048

3. create csr
-> openssl req -new -key kube-controller-manager.key -config kcm.cnf -out kube-controller-manager.csr

4. create crt
-> openssl x509 -req -in kube-controller-manager.csr -CA ca.crt -CAkey ca.key -CAcreateserial -days 365 -extfile kcm.cnf -extensions ext -out kube-controller-manager.crt
